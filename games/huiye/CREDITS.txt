  jump.wav  = "jump.flac"（作者 mieki256，CC0）0.068s，真实跳跃声

    ffmpeg -i jump.flac -c:a pcm_s16le -ac 1 -ar 44100 jump.wav

  hit.wav   = "sword.1.ogg"（攻击挥砍）＋ "sword_clash.1.ogg"（金属撞击），作者 StarNinjas，CC0
